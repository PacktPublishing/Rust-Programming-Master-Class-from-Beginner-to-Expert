// -------------------------------------------
// 			- The Drop Trait
// -------------------------------------------
use std::mem;
use std::rc::Rc;

struct Droplet {
    name: String,
}

impl Drop for Droplet {
    fn drop(&mut self) {
        println!("-> dropping Droplet [{}]", self.name);
    }
}

struct Container {
    first: Droplet,
    second: Droplet,
}

impl Drop for Container {
    fn drop(&mut self) {
        println!("Container::drop body running (fields not dropped yet)");
    }
}

// Point 6: Copy and Drop implemenation for a type is not allowed

fn main() {
    {
        let a = Droplet {
            name: String::from("a"),
        };
        let b = Droplet {
            name: String::from("b"),
        };
        let c = Droplet {
            name: String::from("c"),
        };
        println!("a, b, c created, still inside the block");
    }
    // Point 1: Local variables are dropped in reverse order
    println!("---");

    let d = Droplet {
        name: String::from("d"),
    };
    // d.drop();
    // Point 2: Explicit call to drop is not allowed

    mem::drop(d); // equivalent to: drop(d);
    println!("after mem::drop(d)");
    // Point 3: The drop in std is used to drop a value

    println!("---");

    {
        let cont = Container {
            first: Droplet {
                name: String::from("first"),
            },
            second: Droplet {
                name: String::from("second"),
            },
        };
        println!("Container created");
    }
    // Point 4: Drop order in custom defined types: source code or declaration order

    println!("---");

    {
        let cont = Container {
            first: Droplet {
                name: String::from("L-first"),
            },
            second: Droplet {
                name: String::from("L-second"),
            },
        };
        println!("LoggedContainer created");
    }
    // Point 5: Drop implementation for type, does not affect the source code drop order of its fields

    println!("---");

    {
        let v = vec![
            Droplet {
                name: String::from("v0"),
            },
            Droplet {
                name: String::from("v1"),
            },
            Droplet {
                name: String::from("v2"),
            },
        ];
        println!("vector of Droplets created");
    }

    println!("---");
    //  Point 7: Vectors also follow declaration order in dropping values

    {
        let r1 = Rc::new(Droplet {
            name: String::from("shared"),
        });

        let r2 = Rc::clone(&r1);

        drop(r2); 

        // println!("strong count after dropping r2: {}", Rc::strong_count(&r2)); // add later
        println!("strong count after dropping r2: {}", Rc::strong_count(&r1));
    }
    //  Point 8: Resource to which Rc pointers are pointing are only dropped when strong count is 0
    println!("---");

    let e = Droplet {
        name: String::from("panic-guard"),
    };
    println!("about to panic, watch panic-guard drop during unwind");
    panic!("simulated failure");
    println!("main is ending, remaining locals drop now"); // add later 
    // Point 9: Drop still runs while a panic unwinds
}

/*
-------------------------------------------------------------------------------------------------
Concept/Topic            | Explanation
-------------------------|-----------------------------------------------------------------------
Local variables          | They are dropped in the reverse order of their declaration.

Explicit .drop() call    | It is not allowed.
                         | Use std::mem::drop(value), or simply drop(value), instead.

Struct fields            | Dropped in the declaration order or source code order.

Struct with a Drop impl  | The custom drop() body runs first, before any field is touched.

Vec<T>                   | Elements drop front to back, in source code order

Rc<T> / Arc<T>           | Multiple owners exist.
                         | The original resources is only cleaned when all the strong count is 0

Copy + Drop              | A type can never implement both traits at the same time.

Panic / unwinding        | Drop still runs for locals already in scope variables.
-------------------------------------------------------------------------------------------------
*/
